#include <iostream>
#include <string>
using namespace std;

struct Node {
    int data;
    Node* next;
    Node(int x) : data(x), next(nullptr) {}
};

Node* front_ = nullptr;
Node* rear_ = nullptr;

void enqueue(int x) {
    Node* n = new Node(x);
    if (!rear_) front_ = rear_ = n;
    else {
        rear_->next = n;
        rear_ = n;
    }
}

void dequeue() {
    if (!front_) return;
    Node* t = front_;
    front_ = front_->next;
    if (!front_) rear_ = nullptr;
    delete t;
}

void printQueue() {
    if (!front_) {
        cout << "EMPTY\n";
        return;
    }
    for (Node* c = front_; c; c = c->next) {
        cout << c->data;
        if (c->next) cout << ' ';
    }
    cout << '\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        while (front_) dequeue();
        int q;
        cin >> q;
        while (q--) {
            string op;
            cin >> op;
            if (op == "ENQUEUE") {
                int x;
                cin >> x;
                enqueue(x);
            } else if (op == "DEQUEUE") {
                dequeue();
            } else {
                printQueue();
            }
        }
    }
    return 0;
}





#include <iostream>
using namespace std;

struct Node {
    int key;
    Node* left;
    Node* right;
    Node(int k) : key(k), left(nullptr), right(nullptr) {}
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    Node* root = nullptr;
    for (int i = 0; i < n; i++) {
        int k;
        cin >> k;
        Node* nn = new Node(k);
        if (!root) {
            root = nn;
            continue;
        }
        Node* c = root;
        while (true) {
            if (k < c->key) {
                if (!c->left) { c->left = nn; break; }
                c = c->left;
            } else {
                if (!c->right) { c->right = nn; break; }
                c = c->right;
            }
        }
    }
    int x;
    cin >> x;
    Node* c = root;
    bool found = false, first = true;
    while (c) {
        if (!first) cout << ' ';
        cout << c->key;
        first = false;
        if (x == c->key) { found = true; break; }
        c = x < c->key ? c->left : c->right;
    }
    cout << '\n' << (found ? "FOUND" : "NOT FOUND") << '\n';
    return 0;
}
